(async () => {
  if (location.origin !== 'https://smartlead.hallym.ac.kr') return;
  const request = __ASSIGNMENT_REQUEST__;
  const nonce = '__UNITY_NONCE__', page = location.href;
  const send = data => {
    const packet = JSON.stringify({...data, nonce, page, requestId: request.requestId});
    if (typeof window.Unity?.call === 'function') window.Unity.call(packet);
    else if (typeof window.chrome?.webview?.postMessage === 'function') window.chrome.webview.postMessage(packet);
  };
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), 12000);
  try {
    const url = new URL(request.item.url, location.href);
    if (url.origin !== location.origin || url.username || url.password || !/^\/mod\/(assign|ubassign)\/view\.php$/.test(url.pathname)) throw Error('invalid_assignment');
    const response = await fetch(url.href, {credentials: 'same-origin', signal: controller.signal});
    const final = new URL(response.url || url.href);
    if (final.origin !== location.origin || /^\/login(?:[/.]|$)/.test(final.pathname)) throw Error('login_required');
    if (!response.ok) throw Error('request_failed');
    const doc = new DOMParser().parseFromString(await response.text(), 'text/html');
    if (doc.querySelector('body.notloggedin, input[name="logintoken"], form[action*="/login/"]')) throw Error('login_required');
    const root = doc.querySelector('#region-main') || doc.querySelector('[role="main"]');
    if (!root || !root.querySelector('#intro, .submissionstatustable, .submissionstatus')) throw Error('unknown_assignment_layout');
    const content = root.cloneNode(true);
    // Keep the school's assignment content and table markup. Do not import its navigation,
    // executable code, account data, or controls that could submit/change a school record.
    content.querySelectorAll('script, style, link, meta, base, iframe, object, embed, form, input, button, select, textarea, nav, header, footer, .activity-navigation, .completion-info, .activity-header, .tertiary-navigation, .singlebutton, .nav-tabs').forEach(node => node.remove());
    // The local template supplies the assignment title. Remove only the school's
    // matching page heading, preserving headings inside the assignment itself.
    const normalizeTitle = value => (value || '').normalize('NFC').replace(/\s+/g, ' ').trim();
    const assignmentTitle = normalizeTitle(request.item.title);
    if (assignmentTitle) {
      content.querySelectorAll('h1, h2, h3, h4, h5, h6').forEach(heading => {
        if (!heading.closest('#intro, .submissionstatus, .submissionstatustable, .introattachments') &&
            normalizeTitle(heading.textContent) === assignmentTitle) heading.remove();
      });
    }
    const schoolFileUrl = value => {
      if (!value) return null;
      try {
        const resource = new URL(value, final.href);
        if (resource.origin !== location.origin || resource.username || resource.password ||
            !/^\/(pluginfile|draftfile)\.php\//.test(resource.pathname)) return null;
        return resource.href;
      } catch { return null; }
    };
    const inlineFileUrl = source => {
      const resource = new URL(source);
      if (resource.searchParams.has('forcedownload')) resource.searchParams.delete('forcedownload');
      return resource.href;
    };
    const videoExtension = /\.(mp4|m4v|webm|ogv|ogg|mov)$/i;
    const imageExtension = /\.(apng|png|jpe?g|gif|webp|bmp|avif)$/i;
    const inlineImage = /^data:image\/(apng|png|jpeg|gif|webp|bmp|avif);base64,/i;
    for (const image of content.querySelectorAll('img[data-src]')) {
      const source = schoolFileUrl(image.getAttribute('data-src'));
      if (source) image.setAttribute('src', source);
    }
    // Some Moodle editors store a movie as a link (or just a URL) and rely on
    // the school's scripts to build a player. Recreate it without those scripts.
    const walker = doc.createTreeWalker(content, 4);
    const textNodes = [];
    while (walker.nextNode()) textNodes.push(walker.currentNode);
    for (const node of textNodes) {
      if (node.parentElement.closest('a, video, audio, pre, code')) continue;
      const value = node.textContent.trim();
      if (!/^https:\/\/\S+$/i.test(value)) continue;
      const source = schoolFileUrl(value);
      if (!source || !videoExtension.test(new URL(source).pathname)) continue;
      const link = doc.createElement('a');
      link.href = value; link.textContent = value;
      node.replaceWith(link);
    }
    for (const link of content.querySelectorAll('a[href]')) {
      const source = schoolFileUrl(link.getAttribute('href'));
      if (!source || link.closest('video, audio')) continue;
      const pathname = new URL(source).pathname;
      const inlineSource = inlineFileUrl(source);
      if (videoExtension.test(pathname)) {
        const player = doc.createElement('video');
        player.src = inlineSource;
        link.before(player);
        link.href = inlineSource;
        link.textContent = '영상 파일 열기 · ' + (link.textContent.trim() || '동영상');
      } else if (imageExtension.test(pathname)) {
        // Moodle's attachment links can contain a file-type icon in place of the photo.
        const linkedImages = [...link.querySelectorAll('img')];
        let preview = linkedImages.find(image => {
          const imageSource = schoolFileUrl(image.getAttribute('src'));
          return imageSource && inlineFileUrl(imageSource) === inlineSource;
        });
        if (!preview) {
          preview = doc.createElement('img');
          preview.alt = link.textContent.trim() || linkedImages[0]?.alt || '';
          for (const image of linkedImages) {
            if (!schoolFileUrl(image.getAttribute('src'))) image.remove();
          }
          if (!link.textContent.trim() && !link.querySelector('img')) link.textContent = preview.alt || '이미지 파일 열기';
          link.before(preview);
        }
        preview.setAttribute('src', source);
        preview.classList.add('attachment-preview');
        link.href = source;
      }
    }
    const allowedTags = new Set('DIV SECTION ARTICLE MAIN P SPAN B STRONG I EM U S BR HR H1 H2 H3 H4 H5 H6 UL OL LI TABLE THEAD TBODY TFOOT TR TH TD CAPTION COLGROUP COL A IMG VIDEO AUDIO SOURCE FIGURE FIGCAPTION BLOCKQUOTE PRE CODE DL DT DD SUP SUB SMALL DETAILS SUMMARY'.split(' '));
    for (const node of [...content.querySelectorAll('*')]) {
      if (!allowedTags.has(node.tagName)) { node.replaceWith(...node.childNodes); continue; }
      for (const attribute of [...node.attributes]) {
        if (attribute.name === 'poster' && node.tagName === 'VIDEO') {
          const source = schoolFileUrl(attribute.value);
          if (source) node.setAttribute('poster', source); else node.removeAttribute('poster');
          continue;
        }
        if (attribute.name === 'type' && node.tagName === 'SOURCE') continue;
        if (!['id', 'class', 'style', 'colspan', 'rowspan', 'alt', 'title', 'href', 'src', 'width', 'height', 'open'].includes(attribute.name.toLowerCase())) node.removeAttribute(attribute.name);
      }
      for (const name of ['href', 'src']) {
        const value = node.getAttribute(name);
        if (!value) continue;
        if (name === 'src' && ['VIDEO', 'AUDIO', 'SOURCE'].includes(node.tagName)) {
          const source = schoolFileUrl(value);
          if (source) node.setAttribute(name, inlineFileUrl(source)); else node.removeAttribute(name);
          continue;
        }
        if (name === 'src' && node.tagName === 'IMG' && inlineImage.test(value)) continue;
        const schoolImage = name === 'src' && node.tagName === 'IMG' && schoolFileUrl(value);
        if (schoolImage) { node.setAttribute(name, schoolImage); continue; }
        try {
          const resource = new URL(value, final.href);
          if (resource.protocol !== 'https:' || resource.username || resource.password ||
              (name === 'href' && (resource.origin !== location.origin || !/\/(pluginfile|draftfile)\.php(?:\/|$)/.test(resource.pathname)))) throw Error('unsafe_resource');
          node.setAttribute(name, resource.href);
        } catch { node.removeAttribute(name); }
      }
    }
    for (const player of content.querySelectorAll('video, audio')) {
      const source = player.getAttribute('src') || player.querySelector('source[src]')?.getAttribute('src');
      if (!source) { player.remove(); continue; }
      player.setAttribute('controls', '');
      player.setAttribute('playsinline', '');
      player.setAttribute('preload', player.tagName === 'VIDEO' ? 'metadata' : 'none');
      player.removeAttribute('style');
      player.removeAttribute('width'); player.removeAttribute('height');
      if (player.nextElementSibling?.getAttribute('href') !== source) {
        const link = doc.createElement('a');
        link.href = source; link.textContent = player.tagName === 'VIDEO' ? '영상 파일 열기' : '음원 파일 열기';
        player.after(link);
      }
    }
    // The renderer is a local app document. Embed protected school images using
    // the authenticated collector so they do not depend on a second WebView's cookies.
    const previews = [...content.querySelectorAll('img[src], video[poster]')].flatMap(node => {
      const attribute = node.tagName === 'IMG' ? 'src' : 'poster';
      const source = schoolFileUrl(node.getAttribute(attribute));
      return source ? [{node, attribute, source}] : [];
    });
    // Leave room for the rest of the document and base64 expansion in the native WebView.
    const previewLimit = Math.min(256 * 1024, Math.floor(960 * 1024 / Math.max(1, previews.length)));
    async function rasterMime(blob) {
      const bytes = new Uint8Array(await blob.slice(0, 16).arrayBuffer());
      const header = String.fromCharCode(...bytes);
      if (header.startsWith('\x89PNG\r\n\x1a\n')) return 'image/png';
      if (header.startsWith('\xff\xd8\xff')) return 'image/jpeg';
      if (header.startsWith('GIF87a') || header.startsWith('GIF89a')) return 'image/gif';
      if (header.startsWith('RIFF') && header.startsWith('WEBP', 8)) return 'image/webp';
      if (header.startsWith('BM')) return 'image/bmp';
      if (header.startsWith('ftypavif', 4) || header.startsWith('ftypavis', 4)) return 'image/avif';
      return '';
    }
    async function resizePreview(blob) {
      const bitmap = await createImageBitmap(blob);
      try {
        const canvas = doc.createElement('canvas');
        const context = canvas.getContext('2d');
        if (!context) return null;
        for (const [edge, quality] of [[1000, 0.82], [640, 0.68], [360, 0.58]]) {
          const scale = Math.min(1, edge / Math.max(bitmap.width, bitmap.height));
          canvas.width = Math.max(1, Math.round(bitmap.width * scale));
          canvas.height = Math.max(1, Math.round(bitmap.height * scale));
          context.drawImage(bitmap, 0, 0, canvas.width, canvas.height);
          const preview = await new Promise(resolve => canvas.toBlob(resolve, 'image/webp', quality));
          if (preview && preview.size <= previewLimit) return preview;
        }
        return null;
      } finally { bitmap.close(); }
    }
    async function fetchPreview(source) {
      try {
        const response = await fetch(source, {credentials: 'same-origin', referrer: final.href, signal: controller.signal});
        const imageDownloadLimit = 16 * 1024 * 1024;
        if (!response.ok || Number(response.headers?.get('Content-Length')) > imageDownloadLimit) return null;
        const blob = await response.blob();
        if (!blob.size || blob.size > imageDownloadLimit) return null;
        // The file signature identifies even images sent as generic attachments.
        const mime = await rasterMime(blob);
        const image = mime ? blob.slice(0, blob.size, mime) : blob;
        const preview = mime && image.size <= previewLimit ? image : await resizePreview(image);
        if (!preview) return null;
        return await new Promise((resolve, reject) => { const reader = new FileReader(); reader.onload = () => resolve(reader.result); reader.onerror = reject; reader.readAsDataURL(preview); });
      } catch { return null; }
    }
    let previewsIncomplete = false;
    async function embedPreview({node, attribute, source}) {
      const inlineSource = inlineFileUrl(source);
      let data = null;
      if (!controller.signal.aborted) data = await fetchPreview(source);
      if (!data && inlineSource !== source && !controller.signal.aborted) data = await fetchPreview(inlineSource);
      if (!data) previewsIncomplete = true;
      node.setAttribute(attribute, data || inlineSource);
    }
    for (let offset = 0; offset < previews.length; offset += 3) {
      await Promise.all(previews.slice(offset, offset + 3).map(embedPreview));
    }
    const html = content.innerHTML;
    if (!html || html.length > 4 * 1024 * 1024) throw Error('assignment_too_large');
    const total = Math.ceil(html.length / 8000);
    for (let index = 0; index < total; index++) send({kind: 'assignment_document', url: url.href, index, total, previewsIncomplete, html: html.slice(index * 8000, (index + 1) * 8000)});
  } catch (error) {
    send({kind: 'assignment_document_error', error: error.message === 'login_required' ? '학교 로그인이 만료되었습니다. 계정에서 다시 로그인해 주세요.' :
      '과제 내용을 불러오지 못했습니다. 과제 상세를 다시 열어 주세요.'});
  } finally { clearTimeout(timer); }
})();
